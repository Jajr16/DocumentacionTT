\begin{TestCase}{CP08}{Consultar lista de alumnos inscritos a un ETS}
	\UCitem{Caso de uso relacionado}{\UCref{CU-08}{Consultar lista de alumnos inscritos a un ETS}}
	\UCitem{Descripción}{Validar la consulta y visualización de alumnos inscritos en ETS, incluyendo rendimiento y manejo de datos.}
	\UCitem{Cantidad de ejecuciones}{100}
	\UCitem{Realizado por}{Luis Antonio Flores Esquivel}
	
	\UCitem{Precondiciones}{
		\begin{Citemize}
			\item Docente autenticado
			\item Mínimo 1 ETS activo
			\item Alumnos inscritos (5-100 casos)
			\item Conexión a BD estable
		\end{Citemize}
	}
	
	\UCitem{Conjuntos de Datos}{
		\begin{Citemize}
			\item Listados de prueba con:
			\begin{Citemize}
				\item 5 alumnos (caso mínimo)
				\item 50 alumnos (caso promedio)
				\item 100 alumnos (caso máximo)
			\end{Citemize}
		\end{Citemize}
	}
	
	\UCitem{Procedimiento}{
		\begin{enumerate}[noitemsep,topsep=0pt]
			\item Acceder como docente
			\item Navegar a módulo ETS
			\item Seleccionar ETS específico
			\item Solicitar lista de alumnos
			\item Verificar resultados
		\end{enumerate}
	}
	
	\UCitem{Criterios de Éxito}{
		\begin{Citemize}
			\item Tiempo respuesta <1s (5 alumnos)
			\item Tiempo respuesta <3s (100 alumnos)
			\item Ordenamiento por apellido paterno
			\item Campos completos:
			\begin{Citemize}
				\item Nombre completo
				\item Boleta
				\item Foto
				\item Estatus
			\end{Citemize}
		\end{Citemize}
	}
	
	\UCitem{Pruebas de Estrés}{
		\begin{Citemize}
			\item 1000 solicitudes concurrentes
			\item Tiempo promedio: 2.8s
			\item Sin pérdida de datos
		\end{Citemize}
	}
	
	\UCitem{Validaciones}{
		\begin{Citemize}
			\item Consistencia BD-Interfaz: 100\%
			\item Precisión datos: 100\%
			\item Paginación correcta (>25 registros)
		\end{Citemize}
	}
	
	\UCitem{Estado}{Aprobado}
\end{TestCase}
