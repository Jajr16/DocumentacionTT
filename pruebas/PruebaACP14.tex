%\subsection{Caso de prueba CP02 - Consultar calendario escolar}
\begin{TestCase}{CP02}{Consultar calendario escolar}
	\UCitem{Caso de uso relacionado}{\UCref{CU-02}{Consultar calendario escolar}}
	\UCitem{Descripción}{Validar la funcionalidad de obtención y visualización del calendario escolar del IPN, así como el cálculo de días faltantes para periodos de ETS.}
	\UCitem{Cantidad de ejecuciones}{100}
	\UCitem{Realizado por}{Daniel Martín Huertas Ramírez}
	
	\UCitem{Supuestos y Condiciones Previas}{
		\begin{Citemize}
			\item El usuario inició sesión en el sistema.
			\item El usuario ingresó a la pantalla \IUref{IU02}{Pantalla Consultar calendario escolar}.
			\item Conexión a internet estable.
		\end{Citemize}
	}
	
	\UCitem{Datos de Prueba}{
		\begin{Citemize}
			\item Ninguno requerido.
		\end{Citemize}
	}
	
	\UCitem{Pasos a Ejecutar}{
		\begin{enumerate}[noitemsep,topsep=0pt]
			\item Presionar el ícono de calendario en el menú inferior.
			\item Seleccionar 'Calcular días faltantes para ETS'.
		\end{enumerate}
	}
	
	\UCitem{Resultado Esperado}{
		\begin{Citemize}
			\item Muestra correctamente la imagen del calendario escolar actual.
			\item Proporciona mensaje preciso sobre días faltantes/transcurridos para ETS.
		\end{Citemize}
	}
	
	\UCitem{Errores Detectados}{
		\begin{Citemize}
			\item El sistema no identificaba cuando el usuario estaba en periodo ETS activo.
			\item No reconocía múltiples periodos ETS en un ciclo escolar.
		\end{Citemize}
	}
	
	\UCitem{Correcciones Implementadas}{
		\begin{Citemize}
			\item Ajuste en la lógica para detectar periodos ETS activos.
			\item Modificación en consultas para reconocer múltiples periodos.
		\end{Citemize}
	}
	
	\UCitem{Estado}{Aprobado}
\end{TestCase}
