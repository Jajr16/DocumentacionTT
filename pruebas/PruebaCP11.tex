\begin{TestCase}{CP11}{Mostrar la foto e información del alumno}
	\UCitem{Caso de uso relacionado}{\UCref{CU-11}{Mostrar la foto e información del alumno}}
	\UCitem{Descripción}{Validar la visualización de reportes de alumnos por parte de docentes, incluyendo datos personales y fotografía.}
	\UCitem{Cantidad de ejecuciones}{100}
	\UCitem{Realizado por}{Alejandra De la cruz De la cruz}
	
	\UCitem{Precondiciones}{
		\begin{Citemize}
			\item Docente autenticado en el sistema
			\item ETS activo con alumnos inscritos
			\item Reportes existentes (propios y de otros docentes)
		\end{Citemize}
	}
	
	\UCitem{Conjuntos de Datos}{
		\begin{Citemize}
			\item 5 reportes de prueba con:
			\begin{Citemize}
				\item Fotografías en distintos formatos (JPG/PNG)
				\item Datos de alumnos variables
				\item Marcas temporales recientes e históricas
			\end{Citemize}
		\end{Citemize}
	}
	
	\UCitem{Procedimiento}{
		\begin{enumerate}[noitemsep,topsep=0pt]
			\item Acceder como docente
			\item Navegar a módulo ETS
			\item Seleccionar ETS activo
			\item Acceder a lista de alumnos
			\item Seleccionar reporte específico
			\item Verificar datos y fotografía
		\end{enumerate}
	}
	
	\UCitem{Criterios de Validación}{
		\begin{Citemize}
			\item Visualización completa de:
			\begin{Citemize}
				\item Información del alumno
				\item Fotografía legible
				\item Datos del reporte
			\end{Citemize}
			\item Formato correcto de fechas (DD/MM/AAAA HH:MM)
			\item Tiempo de carga < 1.5 segundos
		\end{Citemize}
	}
	
	\UCitem{Incidentes Resueltos}{
		\begin{Citemize}
			\item \textbf{IS-215}: Formato incorrecto de fecha/hora
			\begin{Citemize}
				\item \textbf{Causa}: Conversión errónea de tipos de datos
				\item \textbf{Solución}: Implementación de formateo estándar
				\item \textbf{Verificación}: Pruebas con 20 zonas horarias
			\end{Citemize}
		\end{Citemize}
	}
	
	\UCitem{Métricas de Rendimiento}{
		\begin{Citemize}
			\item Precisión datos: 100\%
			\item Tiempo promedio respuesta: 1.2s
			\item Uso memoria: <15MB por consulta
		\end{Citemize}
	}
	
	\UCitem{Estado}{Aprobado}
\end{TestCase}
