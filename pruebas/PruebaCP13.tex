\begin{TestCase}{CP13}{Consultar alumnos mediante búsqueda por boleta}
	\UCitem{Caso de uso relacionado}{\UCref{CU-13}{Consultar alumnos mediante búsqueda por boleta}}
	\UCitem{Descripción}{Validar la consulta de alumnos inscritos en ETS mediante búsqueda por número de boleta.}
	\UCitem{Cantidad de ejecuciones}{100}
	\UCitem{Realizado por}{José Alfredo Jiménez Rodríguez}
	
	\UCitem{Requisitos Previos}{
		\begin{Citemize}
			\item Sesión activa de personal de seguridad
			\item Alumnos inscritos en ETS para la fecha
		\end{Citemize}
	}
	
	\UCitem{Datos de Validación}{
		\begin{Citemize}
			\item Boleta válida de alumno inscrito
			\item Fecha de ETS activa
			\item Caso negativo: Boleta no registrada
		\end{Citemize}
	}
	
	\UCitem{Procedimiento}{
		\begin{enumerate}[noitemsep,topsep=0pt]
			\item Autenticarse como personal de seguridad
			\item Acceder a "Consultar Alumnos"
			\item Ingresar número de boleta
			\item Ejecutar consulta
		\end{enumerate}
	}
	
	\UCitem{Criterios de Éxito}{
		\begin{Citemize}
			\item Resultados precisos para búsqueda exacta
			\item Cero resultados para boletas no válidas
			\item Tiempo de respuesta < 2 segundos
		\end{Citemize}
	}
	
	\UCitem{Incidentes Reportados}{
		\begin{Citemize}
			\item \textbf{IS-142}: No retornaba resultados
			\begin{Citemize}
				\item \textbf{Causa}: Error en conexión a BD
				\item \textbf{Solución}: Revisión de timeout de conexión
			\end{Citemize}
		\end{Citemize}
	}
	
	\UCitem{Verificación Post-Corrección}{
		\begin{Citemize}
			\item Pruebas con 50 boletas diferentes
			\item Validación de resultados exactos
			\item Prueba de carga con 100 solicitudes
		\end{Citemize}
	}
	
	\UCitem{Estado}{Aprobado}
\end{TestCase}
