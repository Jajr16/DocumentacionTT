\begin{TestCase}{CP06}{Mostrar información de los ETS asignados}
	\UCitem{Caso de uso relacionado}{\UCref{CU-06}{Mostrar información de los ETS asignados}}
	\UCitem{Descripción}{Validar la visualización estable de datos completos de ETS, incluyendo manejo de concurrencia y validación de datos.}
	\UCitem{Cantidad de ejecuciones}{100}
	\UCitem{Realizado por}{Luis Antonio Flores Esquivel}
	
	\UCitem{Precondiciones}{
		\begin{Citemize}
			\item Docente autenticado
			\item Mínimo 3 ETS asignados
			\item Datos completos en BD:
			\begin{Citemize}
				\item Tipo ETS
				\item Unidad de aprendizaje
				\item Detalles de horario
				\item Asignación de salones
			\end{Citemize}
		\end{Citemize}
	}
	
	\UCitem{Conjunto de Datos}{
		\begin{Citemize}
			\item 5 perfiles de ETS diferentes:
			\begin{Citemize}
				\item Regular (teórico)
				\item Laboratorio
				\item Examen
				\item Extraordinario
				\item Especial
			\end{Citemize}
		\end{Citemize}
	}
	
	\UCitem{Procedimiento de Prueba}{
		\begin{enumerate}[noitemsep,topsep=0pt]
			\item Acceder como docente
			\item Navegar a módulo ETS
			\item Realizar 3 operaciones:
			\begin{enumerate}[noitemsep,topsep=0pt,label=\alph*)]
				\item Visualización normal (1 ETS)
				\item Cambio rápido entre ETS (stress test)
				\item Consulta en baja conexión (<1Mbps)
			\end{enumerate}
		\end{enumerate}
	}
	
	\UCitem{Requisitos de Visualización}{
		\begin{Citemize}
			\item Campos obligatorios:
			\begin{Citemize}
				\item Tipo ETS
				\item Unidad aprendizaje
				\item Periodo/fecha
				\item Detalles de salón
				\item Cupo/disponibilidad
			\end{Citemize}
			\item Tiempo carga <1.5s
		\end{Citemize}
	}
	
	\UCitem{Incidentes Críticos}{
		\begin{Citemize}
			\item \textbf{IS-506}: Crash en cambios rápidos
			\begin{Citemize}
				\item Causa: Race condition en carga de datos
				\item Solución: Implementación de patrón Promise.allSettled()
				\item Verificación: 1000 cambios consecutivos exitosos
			\end{Citemize}
		\end{Citemize}
	}
	
	\UCitem{Métricas de Rendimiento}{
		\begin{Citemize}
			\item Tiempo promedio carga: 1.2s
			\item Uso memoria: <50MB
			\item Estabilidad: 100\% en 100 ejecuciones
		\end{Citemize}
	}
	
	\UCitem{Validaciones Adicionales}{
		\begin{Citemize}
			\item Consistencia BD-UI: 100\%
			\item Prueba offline (caché)
			\item Responsive (mobile/desktop)
		\end{Citemize}
	}
	
	\UCitem{Estado}{Aprobado}
\end{TestCase}
